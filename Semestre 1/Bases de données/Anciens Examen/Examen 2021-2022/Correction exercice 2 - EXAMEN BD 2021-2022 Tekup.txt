Requêtes SQL
-----------------
1.
Select adresse, n°logement, COUNT(*) as nbr-Individus
from Individu I, Location L, Logement LO
where I.N°identité=L.N°identité and L.n°Logement=LO.n°Logement
Group by n°logement 
Having Count(*)>=100;
-----------------------------------
2.
Select N°Logement, TypeLogement
from Logement 
where Loyer > ALL (select Loyer FROM 
Individu I, Location L, Logement LO
where I.N°identité=L.N°identité and L.n°Logement=LO.n°Logement and (NOM = " BEN Brahim" and Prénom ="Mohamed Ali") Or (Nom="Triki" and Prénom="Salma")
);
----------------------------------
3.
select *
from Logement 
where Loyer = (Select Loyer from
Logement L, Individu I , Location LO
where I.N°identité=L.N°identité 
and L.n°Logement=LO.n°Logement and n°Identité= "Id010");
------------------------------------------
4.
Select Nom 
from
Individu I, Location L, Logement LO, Quartier Q
where I.N°identité=L.N°identité 
and L.n°Logement=LO.n°Logement 
and Q.id_quartier=LO.id_quartier
and Nom_quartier like "XX";
