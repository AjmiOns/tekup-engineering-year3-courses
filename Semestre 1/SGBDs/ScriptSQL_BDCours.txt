CREATE TABLE Client(
numClient NUMBER(8) CONSTRAINT PK_Client PRIMARY KEY,
nomClient VARCHAR2(30),
prenomClient VARCHAR2(30),
dat_Naiss DATE,
adresse VARCHAR2(50));

CREATE TABLE Commande(
numCommande NUMBER(4) CONSTRAINT PK_Commande PRIMARY KEY,
datCde DATE DEFAULT SYSDATE,
numClient NUMBER(8),
CONSTRAINT FK_Commande_Client FOREIGN KEY(numClient) REFERENCES Client(numClient));

CREATE TABLE Article(
numArticle NUMBER(4) CONSTRAINT PK_Article PRIMARY KEY,
designation VARCHAR2(100), 
couleur VARCHAR2(30), 
prixHT NUMBER);

CREATE TABLE LigneCommande(
numCmde NUMBER(4),
numArticle NUMBER(4),
qteCdee NUMBER(3),
CONSTRAINT PK_LigneCommande PRIMARY KEY(numCmde,numArticle),
CONSTRAINT FK_LigneCmde_Commande FOREIGN KEY(numCmde) REFERENCES Commande(numCommande),
CONSTRAINT FK_LigneCmde_Article FOREIGN KEY(numArticle) REFERENCES Article(numArticle));

CREATE TABLE Refus(
numCmde NUMBER(4),
numArticle NUMBER(4),
datRefus DATE DEFAULT SYSDATE,
qteRefusee NUMBER(3),
motif VARCHAR2(200),
CONSTRAINT PK_Refus PRIMARY KEY(numCmde,numArticle,datRefus),
CONSTRAINT FK_Refus_LigneCmde FOREIGN KEY(numCmde,numArticle) REFERENCES LigneCommande(numCmde,numArticle));

--Cr er une vue nomm e InfoCommande qui donne la liste de tous les clients et les d tails de leurs commandes respectives  ventuelles

CREATE OR REPLACE VIEW  InfoCommande ( numero_Client, nom, numero_Commande, numero_Article, designation, quantiteeCommandee, prix_unitaire_TTC, quantiteeRefusee)
AS SELECT a.numClient, a.nomClient, b.numCommande, d.numArticle, c.designation, d.qteCdee, prixHT*1.18, qteRefusee
FROM Client a, Commande b, Article c, LigneCommande d , Refus e
WHERE a.numClient = b.numClient(+)
AND b.numCommande = d.numCmde(+)
AND d.numArticle = c.numArticle(+)
AND d.numCmde = e.numCmde(+)
AND d.numArticle = e.numArticle(+)
ORDER BY a.numClient ASC;
