CREATE TABLE Pilote (
    numP NUMBER(8) CONSTRAINT PK_Pilote PRIMARY KEY,
    nomP VARCHAR2(20) NOT NULL,
    prenomP VARCHAR2(20) NOT NULL,
    ville VARCHAR2(30) NOT NULL,
    salaire NUMBER NOT NULL
);

CREATE TABLE Avion (
    numAv NUMBER(6) CONSTRAINT PK_Avion PRIMARY KEY,
    typeAv VARCHAR2(30) NOT NULL,
    capacite NUMBER(6) NOT NULL,
    localisation VARCHAR2(30) NOT NULL
);

CREATE TABLE Vol (
    numVol NUMBER(4) CONSTRAINT PK_Vol PRIMARY KEY,
    numP NUMBER(8) NOT NULL,
    numAv NUMBER(6) NOT NULL,
    villeDep VARCHAR2(25) NOT NULL,
    villeAr VARCHAR2(25) NOT NULL,
    CONSTRAINT fk_vol_pilote FOREIGN KEY (numP)
        REFERENCES Pilote(numP),
    CONSTRAINT fk_vol_avion FOREIGN KEY (numAv)
        REFERENCES Avion(numAv)
);

--B***Nom, prénom et salaire des pilotes dont le salaire est entre 6200 € et 6500 €. 

SELECT nomP, prenomP, salaire
FROM Pilote
WHERE salaire BETWEEN 6200 AND 6500;

-------------------------------------------------------------------------

--C*** Nom, prénom et salaire des pilotes les mieux payés. 

SELECT nomP, prenomP, salaire
FROM Pilote
WHERE salaire = (SELECT MAX(salaire) FROM Pilote);

-------------------------------------------------------------------------
--D*** Caractéristiques des avions localisés à Paris. 
SELECT numAv, typeAv, capacite, localisation
FROM Avion
WHERE localisation = 'Paris';

-------------------------------------------------------------------------

--E***Caractéristiques (NumAv, TypeAv, Capacite, Localisation) des avions localisés dans la même ville que le pilote 'sami'. 


SELECT a.*
FROM Avion a, Pilote p
WHERE UPPER(a.localisation) = UPPER(p.ville)
  AND LOWER(p.prenomP) = 'sami';
 

--OR 

SELECT numAv, typeAv, capacite, localisation
FROM Avion
WHERE localisation = (
    SELECT ville
    FROM Pilote
    WHERE prenomP = 'Sami'
);
-------------------------------------------------------------------------

--*** les avion qui n'ont pas de vol 

SELECT *

--OR

SELECT *
FROM Avion a
WHERE NOT EXISTS (
    SELECT 1
    FROM Vol v
    WHERE v.numAv = a.numAv
