CREATE TABLE Etudiant (
  numEtu NUMBER(6) CONSTRAINT PK_Etudiant PRIMARY KEY,
  nomEtu VARCHAR2(20) NOT NULL,
  prenomEtu VARCHAR2(20) NOT NULL,
  dateNaiss DATE NOT NULL,
  genre CHAR(1) NOT NULL,
  CHECK (numEtu BETWEEN 0 AND 999999),
  CHECK (genre IN ('M','F'))
);


CREATE TABLE Enseignant (
  numEns NUMBER(6) CONSTRAINT PK_Enseignant PRIMARY KEY,
  nomEns VARCHAR2(30) NOT NULL,
  prenomEns VARCHAR2(30) NOT NULL,
  grade CHAR(3) NOT NULL,
  CHECK (grade IN ('ASS','MCF','PRO'))
);


CREATE TABLE Matiere (
  numMat NUMBER(4) CONSTRAINT PK_Matiere PRIMARY KEY,
  nomMat VARCHAR2(25) NOT NULL,
  coefficient NUMBER(1) NOT NULL,
  numEnsRef NUMBER(6),
  CHECK (coefficient BETWEEN 0 AND 9),
  CONSTRAINT FK_Matiere_Ens FOREIGN KEY (numEnsRef)
      REFERENCES Enseignant(numEns)
);


CREATE TABLE Evaluation (
  numEtuRef NUMBER(6) NOT NULL,
  numMatRef NUMBER(4) NOT NULL,
  note NUMBER(4,2) NOT NULL,
  CONSTRAINT PK_Evaluation PRIMARY KEY (numEtuRef, numMatRef),
  CONSTRAINT FK_Evaluation_Etudiant FOREIGN KEY (numEtuRef)
      REFERENCES Etudiant(numEtu),
  CONSTRAINT FK_Evaluation_Matiere FOREIGN KEY (numMatRef)
      REFERENCES Matiere(numMat),
  CHECK (note BETWEEN 0 AND 20)
);
